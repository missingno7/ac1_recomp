cmake_minimum_required(VERSION 3.22)
project(ac1_d3d9_bridge LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "This standalone target validates the Windows native-D3D9 backend")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(ac1_d3d9_bridge STATIC ac1_d3d9_bridge.cpp)
target_include_directories(ac1_d3d9_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(ac1_d3d9_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(ac1_d3d9_bridge PRIVATE /W4 /permissive-)
else()
  target_compile_options(ac1_d3d9_bridge PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(AC1_D3D9_BUILD_SMOKE "Build a real Direct3DCreate9/GetAdapterCount host smoke test" ON)
if(AC1_D3D9_BUILD_SMOKE)
  enable_testing()
  add_executable(ac1_d3d9_bridge_smoke tests/bridge_smoke.cpp)
  target_link_libraries(ac1_d3d9_bridge_smoke PRIVATE ac1_d3d9_bridge user32)
  add_test(NAME ac1_d3d9_bridge_smoke COMMAND ac1_d3d9_bridge_smoke)
  add_test(NAME ac1_d3d9_provider_relative_rejected
           COMMAND ac1_d3d9_bridge_smoke --provider-relative)
  add_test(NAME ac1_d3d9_provider_missing_does_not_fallback
           COMMAND ac1_d3d9_bridge_smoke --provider-missing)
  add_test(NAME ac1_d3d9_provider_without_factory_rejected
           COMMAND ac1_d3d9_bridge_smoke --provider-no-export)

  set(AC1_D3D9_TEST_PROVIDER "" CACHE FILEPATH "Absolute native d3d9.dll path for real-device tests")
  add_executable(ac1_d3d9_bridge_coverage tests/bridge_coverage.cpp)
  target_link_libraries(ac1_d3d9_bridge_coverage PRIVATE ac1_d3d9_bridge user32)
  foreach(test_target ac1_d3d9_bridge_smoke ac1_d3d9_bridge_coverage)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4 /permissive-)
    else()
      target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endforeach()
  add_test(NAME ac1_d3d9_bridge_coverage COMMAND ac1_d3d9_bridge_coverage)
  add_test(NAME ac1_d3d9_bridge_fake_com COMMAND ac1_d3d9_bridge_smoke --fake-com)
  set_tests_properties(ac1_d3d9_bridge_smoke ac1_d3d9_bridge_coverage PROPERTIES TIMEOUT 60)
  if(AC1_D3D9_TEST_PROVIDER)
    if(NOT IS_ABSOLUTE "${AC1_D3D9_TEST_PROVIDER}")
      message(FATAL_ERROR "AC1_D3D9_TEST_PROVIDER must be an absolute DLL path")
    endif()
    set_tests_properties(ac1_d3d9_bridge_smoke ac1_d3d9_bridge_coverage PROPERTIES
      ENVIRONMENT "AC1_D3D9_LIBRARY=${AC1_D3D9_TEST_PROVIDER}")
  endif()
endif()
