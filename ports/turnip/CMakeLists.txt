cmake_minimum_required(VERSION 3.22)
project(ac1_turnip_support LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Turnip support requires Android arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(ADRENOTOOLS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../work/deps/libadrenotools" CACHE PATH "Pinned source")
add_subdirectory("${ADRENOTOOLS_SOURCE}" adrenotools)
add_executable(ac1_vulkan_probe probe/probe.c)
target_include_directories(ac1_vulkan_probe PRIVATE "${MESA_SOURCE}/include")
target_link_libraries(ac1_vulkan_probe PRIVATE adrenotools dl log)
target_compile_options(ac1_vulkan_probe PRIVATE -Wall -Wextra -Werror)
